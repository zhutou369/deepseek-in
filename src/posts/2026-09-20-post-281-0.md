以下係一個簡單嘅 Python 程式碼範例，演示點樣呼叫 DeepSeek API 進行文本分析：

```python
import requests
import json

API_KEY = "YOUR_DEEPSEEK_API_KEY"
API_ENDPOINT = "https://api.deepseek.com/chat/completions" # DeepSeek API 2024年春季更新版

def analyze_fashion_trends(text_data):
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }
    
    prompt = f"""分析以下文本內容，總結香港下一季（例如：2024秋冬季）潮流服飾中，最可能流行嘅五個時尚元素（例如：顏色、剪裁、材質、風格、圖案），並提供每個元素嘅簡要解釋同潛在爆款單品建議。

以下是原始數據：
{text_data}
"""

    payload = {
        "model": "deepseek-chat", # 或者你選擇嘅其他模型
        "messages": [
            {"role": "user", "content": prompt}
        ],
        "temperature": 0.7,
        "max_tokens": 800
    }

    response = requests.post(API_ENDPOINT, headers=headers, data=json.dumps(payload))
    if response.status_code == 200:
        return response.json()['choices'][0]['message']['content']
    else:
        print(f"API 呼叫失敗，狀態碼：{response.status_code}, 錯誤信息：{response.text}")
        return None
```
