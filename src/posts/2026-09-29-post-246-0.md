```python
import os
from deepseek import Deepseek

client = Deepseek(api_key=os.environ.get("DEEPSEEK_API_KEY"))

def generate_toy_review(toy_details):
    response = client.chat.completions.create(
        model="deepseek-v2", # 指定 DeepSeek v2 模型
        messages=[
            {"role": "system", "content": "你是一位資深玩具評論員，擅長以富有情感和細節的方式撰寫玩具開箱文和評測，特別理解香港本地的玩具文化。請用香港繁體中文撰寫。"},
            {"role": "user", "content": f"請根據以下資訊，幫我撰寫一篇關於香港手辦或潮流玩具的開箱評測文。內容需包含個人情感和細節描述。\n\n資訊：\n{toy_details}"
            }
        ],
        temperature=0.7, # 調整創意程度，0-1之間，越高越有創意
        max_tokens=800 # 控制生成文章的長度
    )
    return response.choices[0].message.content

# 假設這是你從第一步整理出來的「情懷」與評測重點
toy_info = """
玩具名稱：MegaHouse G.E.M.系列 鬼滅之刃 竈門禰豆子 掌心禰豆子 (豪華版)
購買地點：信和中心某樓層嘅玩具店
購買經歷：等咗大半年終於預訂到，開箱嗰陣真係好興奮！
情懷點：Q版造型超可愛，禰豆子坐姿設計好治癒，擺喺檯面工作都開心啲。豪華版嘅水波紋底座好有質感，細節位（髮絲、和服花紋）做得好精緻，完全還原動漫神韻。"""

# 生成文章草稿
review_draft = generate_toy_review(toy_info)
print(review_draft)
```
